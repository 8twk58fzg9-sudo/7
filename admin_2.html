CREATE OR REPLACE FUNCTION is_admin()
RETURNS BOOLEAN
LANGUAGE sql
STABLE
SECURITY INVOKER
SET search_path = ''
AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.admin_users
    WHERE user_id = (select auth.uid())
  );
$$;

CREATE OR REPLACE FUNCTION ensure_order_number()
RETURNS TRIGGER
LANGUAGE plpgsql
SET search_path = ''
AS $$
BEGIN
  IF NULLIF(BTRIM(NEW.order_number), '') IS NULL THEN
    NEW.order_number := 'CT-' || to_char(now(), 'YYYYMMDD') || '-' || upper(substr(md5(random()::text || clock_timestamp()::text), 1, 6));
  END IF;
  RETURN NEW;
END;
$$;
DROP TRIGGER IF EXISTS orders_ensure_order_number ON orders;
CREATE TRIGGER orders_ensure_order_number BEFORE INSERT ON orders FOR EACH ROW EXECUTE FUNCTION ensure_order_number();

CREATE OR REPLACE FUNCTION log_stock_movement()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    INSERT INTO public.stock_movements(product_id, product_name, old_stock, new_stock, delta, reason)
    VALUES (NEW.id, NEW.name, NULL, NEW.stock, NEW.stock, 'product_created');
    RETURN NEW;
  ELSIF TG_OP = 'UPDATE' THEN
    IF NEW.stock IS DISTINCT FROM OLD.stock THEN
      INSERT INTO public.stock_movements(product_id, product_name, old_stock, new_stock, delta, reason)
      VALUES (NEW.id, NEW.name, OLD.stock, NEW.stock, NEW.stock - OLD.stock, 'stock_updated');
    END IF;
    RETURN NEW;
  ELSIF TG_OP = 'DELETE' THEN
    INSERT INTO public.stock_movements(product_id, product_name, old_stock, new_stock, delta, reason)
    VALUES (OLD.id, OLD.name, OLD.stock, 0, -OLD.stock, 'product_deleted_or_sold_out');
    RETURN OLD;
  END IF;
  RETURN NULL;
END;
$$;
DROP TRIGGER IF EXISTS products_log_stock_insert ON products;
CREATE TRIGGER products_log_stock_insert AFTER INSERT ON products FOR EACH ROW EXECUTE FUNCTION log_stock_movement();
DROP TRIGGER IF EXISTS products_log_stock_update ON products;
CREATE TRIGGER products_log_stock_update AFTER UPDATE ON products FOR EACH ROW EXECUTE FUNCTION log_stock_movement();
DROP TRIGGER IF EXISTS products_log_stock_delete ON products;
CREATE TRIGGER products_log_stock_delete AFTER DELETE ON products FOR EACH ROW EXECUTE FUNCTION log_stock_movement();

CREATE OR REPLACE FUNCTION purchase_products(items JSONB)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  item JSONB;
  item_name TEXT;
  item_qty INTEGER;
  current_stock INTEGER;
BEGIN
  IF jsonb_typeof(items) <> 'array' THEN
    RAISE EXCEPTION 'items must be a JSON array';
  END IF;

  FOR item IN SELECT * FROM jsonb_array_elements(items)
  LOOP
    item_name := NULLIF(BTRIM(item->>'name'), '');
    item_qty := COALESCE((item->>'qty')::INTEGER, 1);

    IF item_name IS NULL OR item_qty < 1 THEN
      RAISE EXCEPTION 'Invalid purchase item';
    END IF;

    SELECT stock INTO current_stock
    FROM public.products
    WHERE name = item_name
    FOR UPDATE;

    IF current_stock IS NULL THEN
      RAISE EXCEPTION 'Product not found: %', item_name;
    END IF;

    IF current_stock < item_qty THEN
      RAISE EXCEPTION 'Not enough stock for %', item_name;
    END IF;

    IF current_stock = item_qty THEN
      DELETE FROM public.products WHERE name = item_name;
    ELSE
      UPDATE public.products
      SET stock = current_stock - item_qty,
          status = CASE WHEN current_stock - item_qty <= 0 THEN 'sold' ELSE status END
      WHERE name = item_name;
    END IF;
  END LOOP;
END;
$$;

REVOKE ALL ON FUNCTION purchase_products(JSONB) FROM PUBLIC;
REVOKE EXECUTE ON FUNCTION purchase_products(JSONB) FROM anon, authenticated;
REVOKE EXECUTE ON FUNCTION ensure_order_number() FROM PUBLIC;

CREATE TABLE IF NOT EXISTS rate_limits (
  rate_key TEXT PRIMARY KEY,
  hits INTEGER NOT NULL DEFAULT 1 CHECK (hits > 0),
  window_start TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
ALTER TABLE rate_limits ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON TABLE rate_limits FROM PUBLIC, anon, authenticated;

CREATE OR REPLACE FUNCTION enforce_rate_limit(p_key TEXT, p_max_hits INTEGER, p_window_seconds INTEGER)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  current_hits INTEGER;
BEGIN
  IF p_key IS NULL OR btrim(p_key) = '' OR p_max_hits < 1 OR p_window_seconds < 1 THEN
    RAISE EXCEPTION 'Invalid rate limit config';
  END IF;

  INSERT INTO public.rate_limits(rate_key, hits, window_start, updated_at)
  VALUES (left(p_key, 220), 1, now(), now())
  ON CONFLICT (rate_key) DO UPDATE SET
    hits = CASE
      WHEN public.rate_limits.window_start < now() - make_interval(secs => p_window_seconds) THEN 1
      ELSE public.rate_limits.hits + 1
    END,
    window_start = CASE
      WHEN public.rate_limits.window_start < now() - make_interval(secs => p_window_seconds) THEN now()
      ELSE public.rate_limits.window_start
    END,
    updated_at = now()
  RETURNING hits INTO current_hits;

  IF current_hits > p_max_hits THEN
    RAISE EXCEPTION 'rate limit exceeded' USING ERRCODE = 'P0001';
  END IF;
END;
$$;
REVOKE ALL ON FUNCTION enforce_rate_limit(TEXT, INTEGER, INTEGER) FROM PUBLIC, anon, authenticated;

CREATE OR REPLACE FUNCTION create_order_and_purchase(order_payload JSONB)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  item JSONB;
  item_name TEXT;
  item_qty INTEGER;
  current_stock INTEGER;
  current_price NUMERIC;
  current_status TEXT;
  new_order_id BIGINT;
  order_items JSONB;
  expected_total NUMERIC := 0;
  submitted_total NUMERIC;
  submitted_payment_method TEXT;
  submitted_payment_status TEXT;
  submitted_warehouse_status TEXT;
  submitted_name TEXT;
  submitted_email TEXT;
  submitted_phone TEXT;
  submitted_phone_digits TEXT;
  submitted_address TEXT;
  submitted_delivery TEXT;
  submitted_delivery_key TEXT;
  submitted_note TEXT;
  submitted_order_number TEXT;
  submitted_client_order_key TEXT;
  submitted_service_ids JSONB;
  service_id TEXT;
  service_price NUMERIC;
  service_total NUMERIC := 0;
  delivery_fee NUMERIC := 0;
  delivery_days INTEGER := 0;
  site_settings_json JSONB := '{}'::jsonb;
  delivery_method JSONB;
  existing_order_id BIGINT;
  recent_order_count INTEGER;
  unexpected_key TEXT;
  seen_item_names TEXT[] := ARRAY[]::TEXT[];
  seen_service_ids TEXT[] := ARRAY[]::TEXT[];
BEGIN
  IF jsonb_typeof(order_payload) <> 'object' THEN
    RAISE EXCEPTION 'Invalid order payload';
  END IF;

  SELECT field_name INTO unexpected_key
  FROM jsonb_object_keys(order_payload) AS payload_key(field_name)
  WHERE field_name NOT IN (
    'order_number','client_order_key','customer_name','customer_email','customer_phone',
    'address','delivery_key','delivery','items','service_ids','total','note',
    'terms_accepted','terms_accepted_at','payment_method','payment_status',
    'payment_reference','payment_url','reservation_expires_at',
    'warehouse_sync_status','warehouse_reference'
  )
  LIMIT 1;

  IF unexpected_key IS NOT NULL THEN
    RAISE EXCEPTION 'Unexpected order field: %', unexpected_key;
  END IF;

  order_items := order_payload->'items';
  IF COALESCE(order_payload->>'total', '') !~ '^[0-9]+([.][0-9]{1,2})?$' THEN
    RAISE EXCEPTION 'Invalid order total';
  END IF;
  submitted_total := (order_payload->>'total')::NUMERIC;
  submitted_payment_method := COALESCE(NULLIF(BTRIM(order_payload->>'payment_method'), ''), 'cod');
  -- Public checkout must never be trusted to mark money or warehouse state as finished.
  -- GoPay/webhook/admin code can update these fields later after real verification.
  submitted_payment_status := 'pending';
  submitted_warehouse_status := 'pending';
  submitted_name := NULLIF(BTRIM(order_payload->>'customer_name'), '');
  submitted_email := lower(NULLIF(BTRIM(order_payload->>'customer_email'), ''));
  submitted_phone := NULLIF(BTRIM(order_payload->>'customer_phone'), '');
  submitted_phone_digits := regexp_replace(COALESCE(submitted_phone, ''), '[^0-9]', '', 'g');
  submitted_address := NULLIF(BTRIM(order_payload->>'address'), '');
  submitted_delivery := NULLIF(BTRIM(order_payload->>'delivery'), '');
  submitted_delivery_key := NULLIF(BTRIM(order_payload->>'delivery_key'), '');
  submitted_note := NULLIF(BTRIM(order_payload->>'note'), '');
  submitted_order_number := upper(NULLIF(BTRIM(order_payload->>'order_number'), ''));
  submitted_client_order_key := upper(NULLIF(BTRIM(order_payload->>'client_order_key'), ''));
  submitted_service_ids := COALESCE(order_payload->'service_ids', '[]'::jsonb);

  IF jsonb_typeof(order_items) <> 'array' THEN
    RAISE EXCEPTION 'items must be a JSON array';
  END IF;

  IF jsonb_array_length(order_items) < 1 OR jsonb_array_length(order_items) > 20 THEN
    RAISE EXCEPTION 'Invalid number of items';
  END IF;

  IF submitted_name IS NULL OR char_length(submitted_name) NOT BETWEEN 2 AND 120 THEN
    RAISE EXCEPTION 'Invalid customer name';
  END IF;

  IF submitted_email IS NULL OR submitted_email !~* '^[^@\s]+@[^@\s]+\.[^@\s]+$' THEN
    RAISE EXCEPTION 'Invalid customer email';
  END IF;

  IF submitted_order_number IS NULL
     OR submitted_order_number !~ '^CTX-[0-9]{8}-[A-Z0-9]{5}$'
     OR submitted_client_order_key IS DISTINCT FROM submitted_order_number THEN
    RAISE EXCEPTION 'Invalid order metadata';
  END IF;

  -- Serialise retries for the same client order key. A lost HTTP response can
  -- then be retried safely without creating another order or reducing stock twice.
  PERFORM pg_advisory_xact_lock(hashtext(submitted_order_number));
  SELECT id INTO existing_order_id
  FROM public.orders
  WHERE order_number = submitted_order_number
    AND lower(customer_email) = submitted_email
  LIMIT 1;

  IF existing_order_id IS NOT NULL THEN
    RETURN jsonb_build_object('order_id', existing_order_id, 'idempotent', true);
  END IF;

  IF EXISTS (SELECT 1 FROM public.orders WHERE order_number = submitted_order_number) THEN
    RAISE EXCEPTION 'Duplicate order number';
  END IF;

  -- A global database guard limits automated floods; email/user buckets limit
  -- individual customers. IP-based 429 handling belongs in the Edge Function.
  PERFORM public.enforce_rate_limit('order:global:' || to_char(now(), 'YYYYMMDDHH24MI'), 120, 120);
  PERFORM public.enforce_rate_limit('order:email:' || md5(submitted_email), 5, 600);
  IF auth.uid() IS NOT NULL THEN
    PERFORM public.enforce_rate_limit('order:user:' || md5(auth.uid()::text), 8, 600);
  END IF;

  SELECT count(*) INTO recent_order_count
  FROM public.orders
  WHERE lower(customer_email) = submitted_email
    AND created_at > now() - interval '10 minutes';

  IF recent_order_count >= 5 THEN
    RAISE EXCEPTION 'Too many recent orders for this email';
  END IF;

  IF submitted_phone IS NOT NULL AND char_length(submitted_phone_digits) NOT BETWEEN 9 AND 15 THEN
    RAISE EXCEPTION 'Invalid customer phone';
  END IF;

  SELECT COALESCE(settings, '{}'::jsonb)
    INTO site_settings_json
    FROM public.site_settings
   WHERE key = 'main'
   LIMIT 1;

  SELECT method
    INTO delivery_method
    FROM jsonb_array_elements(COALESCE(site_settings_json->'deliveryMethods', '[]'::jsonb)) AS method
	   WHERE method->>'id' = submitted_delivery_key
	     AND lower(COALESCE(method->>'enabled', 'true')) <> 'false'
	   LIMIT 1;

  IF delivery_method IS NULL THEN
    SELECT method
      INTO delivery_method
      FROM jsonb_array_elements(
        '[{"id":"kurier","name":"Kuriér GLS","price":5.90,"days":2},{"id":"packeta","name":"Packeta","price":3.90,"days":3},{"id":"osobne","name":"Osobný odber","price":0,"days":0}]'::jsonb
      ) AS method
     WHERE method->>'id' = submitted_delivery_key
     LIMIT 1;
  END IF;

  IF delivery_method IS NULL THEN
    RAISE EXCEPTION 'Invalid delivery method';
  END IF;

  delivery_fee := GREATEST(0, LEAST(500, COALESCE((delivery_method->>'price')::NUMERIC, 0)));
  delivery_days := GREATEST(0, LEAST(60, COALESCE((delivery_method->>'days')::INTEGER, 0)));
  submitted_delivery := COALESCE(NULLIF(BTRIM(delivery_method->>'name'), ''), submitted_delivery_key)
    || ' - '
    || CASE WHEN delivery_fee = 0 THEN 'zdarma' ELSE replace(to_char(delivery_fee, 'FM999990.00'), '.', ',') || ' EUR' END
    || ' ('
    || CASE WHEN delivery_days > 0 THEN 'do ' || delivery_days || ' dní' ELSE 'podľa dohody' END
    || ')';

  IF delivery_fee > 0 AND COALESCE(char_length(submitted_address), 0) < 8 THEN
    RAISE EXCEPTION 'Delivery address is required';
  END IF;

  IF submitted_address IS NOT NULL AND char_length(submitted_address) > 300 THEN
    RAISE EXCEPTION 'Invalid delivery address';
  END IF;

  IF submitted_note IS NOT NULL AND char_length(submitted_note) > 1000 THEN
    RAISE EXCEPTION 'Invalid order note';
  END IF;

  IF char_length(COALESCE(order_payload->>'warehouse_reference', '')) > 160 THEN
    RAISE EXCEPTION 'Invalid order metadata';
  END IF;

  IF submitted_total < 0 OR submitted_total > 100000 THEN
    RAISE EXCEPTION 'Invalid order total';
  END IF;

  IF order_payload->>'terms_accepted' IS DISTINCT FROM 'true' THEN
    RAISE EXCEPTION 'Terms must be accepted';
  END IF;

  IF submitted_payment_method NOT IN ('cod','bank_transfer','gopay','stripe') THEN
    RAISE EXCEPTION 'Invalid payment method';
  END IF;

  IF jsonb_typeof(submitted_service_ids) <> 'array'
     OR jsonb_array_length(submitted_service_ids) > 10
     OR EXISTS (
       SELECT 1
       FROM jsonb_array_elements(submitted_service_ids) AS selected_service(value)
       WHERE jsonb_typeof(value) <> 'string'
          OR char_length(btrim(value #>> '{}')) NOT BETWEEN 1 AND 50
     ) THEN
    RAISE EXCEPTION 'Invalid service selection';
  END IF;

  FOR service_id IN SELECT jsonb_array_elements_text(submitted_service_ids)
  LOOP
    IF service_id = ANY(seen_service_ids) THEN
      RAISE EXCEPTION 'Duplicate service selection';
    END IF;
    seen_service_ids := array_append(seen_service_ids, service_id);

    SELECT CASE
      WHEN COALESCE(service->>'price', '') ~ '^[0-9]+([.][0-9]{1,2})?$'
        THEN (service->>'price')::NUMERIC
      ELSE NULL
    END INTO service_price
    FROM public.site_settings AS site
    CROSS JOIN LATERAL jsonb_array_elements(
      CASE
        WHEN jsonb_typeof(site.settings->'services') = 'array' THEN site.settings->'services'
        ELSE '[]'::jsonb
      END
    ) AS service
    WHERE site.key = 'main'
      AND service->>'id' = service_id
      AND COALESCE(service->>'enabled', 'true') = 'true'
    LIMIT 1;

    IF NOT FOUND OR service_price IS NULL OR service_price NOT BETWEEN 0 AND 1000 THEN
      RAISE EXCEPTION 'Invalid service selection';
    END IF;
    service_total := service_total + service_price;
  END LOOP;

  FOR item IN SELECT * FROM jsonb_array_elements(order_items)
  LOOP
    IF jsonb_typeof(item) <> 'object'
       OR EXISTS (
         SELECT 1
         FROM jsonb_object_keys(item) AS item_key(field_name)
         WHERE field_name NOT IN ('name','qty','price')
       ) THEN
      RAISE EXCEPTION 'Unexpected purchase item field';
    END IF;

    item_name := NULLIF(BTRIM(item->>'name'), '');
    IF COALESCE(item->>'qty', '') !~ '^[0-9]+$' THEN
      RAISE EXCEPTION 'Invalid purchase item';
    END IF;
    item_qty := (item->>'qty')::INTEGER;

    IF item_name IS NULL OR char_length(item_name) > 180 OR item_qty < 1 OR item_qty > 20 THEN
      RAISE EXCEPTION 'Invalid purchase item';
    END IF;

    IF lower(item_name) = ANY(seen_item_names) THEN
      RAISE EXCEPTION 'Duplicate purchase item';
    END IF;
    seen_item_names := array_append(seen_item_names, lower(item_name));

    SELECT stock, price, status INTO current_stock, current_price, current_status
    FROM public.products
    WHERE name = item_name
    FOR UPDATE;

    IF current_stock IS NULL THEN
      RAISE EXCEPTION 'Product not found: %', item_name;
    END IF;

    IF current_stock < item_qty THEN
      RAISE EXCEPTION 'Not enough stock for %', item_name;
    END IF;

    IF current_status <> 'active' THEN
      RAISE EXCEPTION 'Product is not available: %', item_name;
    END IF;

    expected_total := expected_total + (current_price * item_qty);
  END LOOP;

  expected_total := expected_total + delivery_fee + service_total;

  IF abs(submitted_total - expected_total) > 0.01 THEN
    RAISE EXCEPTION 'Invalid order total';
  END IF;

  INSERT INTO public.orders (
    order_number,
    customer_name,
    customer_email,
    customer_phone,
    address,
    delivery,
    items,
    total,
    status,
    note,
    terms_accepted,
    terms_accepted_at,
    sync_status,
    payment_method,
    payment_status,
    payment_reference,
    payment_url,
    reservation_expires_at,
    warehouse_sync_status,
    warehouse_reference
  )
  VALUES (
    submitted_order_number,
    submitted_name,
    submitted_email,
    submitted_phone,
    submitted_address,
    submitted_delivery,
    order_items,
    submitted_total,
    'new',
    submitted_note,
    true,
    now(),
    'ok',
    submitted_payment_method,
    submitted_payment_status,
    NULL,
    NULL,
    CASE WHEN submitted_payment_method IN ('gopay','stripe') THEN now() + interval '30 minutes' ELSE NULL END,
    submitted_warehouse_status,
    NULL
  )
  RETURNING id INTO new_order_id;

  FOR item IN SELECT * FROM jsonb_array_elements(order_items)
  LOOP
    item_name := NULLIF(BTRIM(item->>'name'), '');
    item_qty := COALESCE((item->>'qty')::INTEGER, 1);

    SELECT stock INTO current_stock
    FROM public.products
    WHERE name = item_name
    FOR UPDATE;

    IF current_stock = item_qty THEN
      DELETE FROM public.products WHERE name = item_name;
    ELSE
      UPDATE public.products
      SET stock = current_stock - item_qty,
          status = CASE WHEN current_stock - item_qty <= 0 THEN 'sold' ELSE status END
      WHERE name = item_name;
    END IF;
  END LOOP;

  RETURN jsonb_build_object('order_id', new_order_id, 'idempotent', false);
END;
$$;

REVOKE ALL ON FUNCTION create_order_and_purchase(JSONB) FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION create_order_and_purchase(JSONB) TO service_role;

CREATE OR REPLACE FUNCTION track_customer_order(p_order_number TEXT, p_email TEXT, p_request_key TEXT)
RETURNS TABLE (
  order_number TEXT,
  items JSONB,
  total NUMERIC,
  status TEXT,
  payment_method TEXT,
  payment_status TEXT,
  payment_url TEXT,
  invoice_status TEXT,
  invoice_reference TEXT,
  invoice_url TEXT,
  delivery TEXT,
  created_at TIMESTAMPTZ
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  normalized_order_number TEXT := upper(btrim(COALESCE(p_order_number, '')));
  normalized_email TEXT := lower(btrim(COALESCE(p_email, '')));
BEGIN
  IF normalized_order_number !~ '^CTX-[0-9]{8}-[A-Z0-9]{5}$'
     OR char_length(normalized_email) NOT BETWEEN 5 AND 254
     OR normalized_email !~* '^[^@\s]+@[^@\s]+\.[^@\s]+$'
     OR COALESCE(p_request_key, '') !~ '^[a-f0-9]{64}$' THEN
    RAISE EXCEPTION 'Invalid tracking request';
  END IF;

  PERFORM public.enforce_rate_limit('track:global:' || to_char(now(), 'YYYYMMDDHH24MI'), 300, 120);
  PERFORM public.enforce_rate_limit('track:request:' || md5(p_request_key), 20, 60);
  PERFORM public.enforce_rate_limit('track:email:' || md5(normalized_email), 8, 600);

  RETURN QUERY
  SELECT
    o.order_number, o.items, o.total, o.status, o.payment_method, o.payment_status,
    o.payment_url, o.invoice_status, o.invoice_reference, o.invoice_url,
    o.delivery, o.created_at
  FROM public.orders AS o
  WHERE upper(o.order_number) = normalized_order_number
    AND lower(o.customer_email) = normalized_email
  ORDER BY o.created_at DESC
  LIMIT 1;
END;
$$;
REVOKE ALL ON FUNCTION track_customer_order(TEXT, TEXT, TEXT) FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION track_customer_order(TEXT, TEXT, TEXT) TO service_role;
GRANT EXECUTE ON FUNCTION enforce_rate_limit(TEXT, INTEGER, INTEGER) TO service_role;

CREATE OR REPLACE FUNCTION public.admin_autopilot()
RETURNS JSONB
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE
  fixed_zero_stock INTEGER := 0;
  generated_skus INTEGER := 0;
  synced_warehouse_skus INTEGER := 0;
BEGIN
  IF NOT public.is_admin() THEN
    RAISE EXCEPTION 'Admin only';
  END IF;

  UPDATE public.products
  SET status = 'sold'
  WHERE COALESCE(stock, 0) = 0
    AND status IS DISTINCT FROM 'sold';
  GET DIAGNOSTICS fixed_zero_stock = ROW_COUNT;

  UPDATE public.products
  SET sku = 'CTX-PC-' || id::TEXT
  WHERE NULLIF(BTRIM(COALESCE(sku, '')), '') IS NULL;
  GET DIAGNOSTICS generated_skus = ROW_COUNT;

  UPDATE public.products
  SET warehouse_sku = sku
  WHERE NULLIF(BTRIM(COALESCE(warehouse_sku, '')), '') IS NULL
    AND NULLIF(BTRIM(COALESCE(sku, '')), '') IS NOT NULL;
  GET DIAGNOSTICS synced_warehouse_skus = ROW_COUNT;

  INSERT INTO public.admin_events(user_id, action, entity_type, entity_id, summary, metadata)
  VALUES (
    auth.uid(),
    'admin_autopilot',
    'system',
    NULL,
    'Autopilot údržba skladu a SKU',
    jsonb_build_object(
      'fixed_zero_stock', fixed_zero_stock,
      'generated_skus', generated_skus,
      'synced_warehouse_skus', synced_warehouse_skus
    )
  );

  RETURN jsonb_build_object(
    'fixed_zero_stock', fixed_zero_stock,
    'generated_skus', generated_skus,
    'synced_warehouse_skus', synced_warehouse_skus
  );
END;
$$;
